public boolean verify(HttpHeaders headers, String body) {
    String eventId = headers.getFirst("X-Event-Id");
    String eventType = headers.getFirst("X-Event-Type");
    String timestamp = headers.getFirst("X-Event-Timestamp");
    String signature = headers.getFirst("X-Event-Signature");

    // 时间窗校验
    long ts = Long.parseLong(timestamp);
    if (Math.abs(System.currentTimeMillis() / 1000 - ts) > 300) {
        return false;
    }

    String bodyHash = DigestUtils.sha256Hex(body);
    String canonical = eventId + "\n" + eventType + "\n" + timestamp + "\n" + bodyHash;
    String expected = Base64.encodeBase64String(
        HmacUtils.hmacSha256(webhookSecret, canonical));

    return MessageDigest.isEqual(
        expected.getBytes(StandardCharsets.UTF_8),
        signature.getBytes(StandardCharsets.UTF_8));
}
